% CV PDF styling: designed to match the prior Word CV while remaining Quarto-native.
\usepackage{newtxtext}
\usepackage{microtype}
\usepackage{xcolor}
\usepackage{enumitem}
\usepackage[explicit]{titlesec}
\usepackage{fancyhdr}
\usepackage{hyperref}
\usepackage{needspace}
\usepackage{etoolbox}

% Times-like serif via TeX Live.

\definecolor{cvgray}{HTML}{777777}
\hypersetup{colorlinks=true,urlcolor=black,linkcolor=black}
\urlstyle{same}

\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\setlength{\emergencystretch}{2em}
\raggedbottom

% Header / footer.
\pagestyle{fancy}
\fancyhf{}
\fancyhead[R]{\footnotesize\color{cvgray}Updated on \today}
\fancyfoot[C]{\small\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\setlength{\headheight}{14pt}
\setlength{\headsep}{12pt}
\setlength{\footskip}{22pt}

% Section headings: bold, uppercase, and underlined like the Word CV.
\titleformat{\section}
  {\needspace{3\baselineskip}\bfseries\large}
  {}{0pt}{\underline{\MakeUppercase{#1}}}
\titlespacing*{\section}{0pt}{1.05em}{0.48em}
\titleformat{\subsection}
  {\needspace{3\baselineskip}\bfseries\large}
  {}{0pt}{\underline{\MakeUppercase{#1}}}
\titlespacing*{\subsection}{0pt}{1.05em}{0.48em}

% Suppress Quarto/Pandoc title block; cv.lua draws the custom header.
\AtBeginDocument{\let\maketitle\relax}

% Compact bullet lists in education entries.
\setlist[itemize]{
  leftmargin=2.35em,
  labelsep=0.65em,
  itemsep=0.05em,
  topsep=0.15em,
  parsep=0pt,
  partopsep=0pt
}

% Two-column CV row: content left, date right.
\newenvironment{cvrow}[1]{%
  \par\vspace{0.18em}%
  \noindent
  \begin{minipage}[t]{0.79\textwidth}%
  \vspace{0pt}%
}{%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.18\textwidth}%
  \vspace{0pt}\raggedleft\strut\cvdatevalue
  \end{minipage}%
  \par\vspace{0.56em}%
}
\newcommand{\cvdatevalue}{}
\newcommand{\cvrowstart}[1]{\renewcommand{\cvdatevalue}{#1}\begin{cvrow}{#1}}
\newcommand{\cvrowend}{\end{cvrow}}

% Hanging-indent entries for publications and talks.
\newenvironment{cvpub}{%
  \par\noindent\hangindent=0.65em\hangafter=1
}{%
  \par\vspace{0.36em}
}

% Custom centered name/contact block.
\newcommand{\cvname}[1]{%
  \begin{center}
    {\fontsize{19}{21}\selectfont\bfseries #1}\par
  \end{center}
  \vspace{0.5em}
}
\newcommand{\cvcontact}[1]{%
  \begin{center}
    {\normalsize #1}\par
  \end{center}
  \vspace{1.2em}
}
